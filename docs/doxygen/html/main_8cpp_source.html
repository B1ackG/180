73bool checkModbusConnection(QSplashScreen* splash, const QString& ip, int port, const QString& deviceName, int timeoutMs = 5000)
74{
75 QString message = QString("正在连接 %1 (%2:%3)...").arg(deviceName).arg(ip).arg(port);
76 splash->showMessage(message, Qt::AlignBottom | Qt::AlignCenter, Qt::white);
77 qApp->processEvents();
78
79 QElapsedTimer timer;
80 timer.start();
81
82 try {
83 // 自检阶段使用独立 TCP 探测，避免干扰运行期 Modbus 管理器连接状态
84 QTcpSocket socket;
85 socket.connectToHost(ip, static_cast<quint16>(port));
86 bool connectionSuccess = socket.waitForConnected(timeoutMs);
87
88 if (connectionSuccess) {
89 QString successMsg = QString("%1 连接成功 (%2ms)").arg(deviceName).arg(timer.elapsed());
90 splash->showMessage(successMsg, Qt::AlignBottom | Qt::AlignCenter, Qt::green);
91 qApp->processEvents();
92 socket.disconnectFromHost();
93 return true;
94 } else {
95 QString failMsg = QString("%1 连接失败或超时").arg(deviceName);
96 splash->showMessage(failMsg, Qt::AlignBottom | Qt::AlignCenter, Qt::yellow);
97 qApp->processEvents();
98 return false;
99 }
100
101 } catch (const std::exception& e) {
102 qDebug() << QString("%1连接异常: %2").arg(deviceName).arg(e.what());
103 return false;
104 } catch (...) {
105 qDebug() << QString("%1连接未知异常").arg(deviceName);
106 return false;
107 }
108}
