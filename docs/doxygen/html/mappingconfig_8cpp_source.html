191QString MappingConfig::mapControlName(const QString &objectName) const
192{
193 // 先查找精确匹配
194 if (m_controlNameMap.contains(objectName)) {
195 return m_controlNameMap[objectName];
196 }
197
198 // 特殊情况：包含 "steeringModeSelector" 的映射
199 if (objectName.contains("steeringModeSelector")) {
200 return "转向模式选择器";
201 }
202
203 // 如果没有精确匹配，尝试部分匹配（按前缀）
204 // 例如：所有 VeSupSec 开头的控件都显示为"回转升降"
205 if (objectName.startsWith("VeSupSec_")) {
206 return QString("回转升降模块-%1").arg(objectName);
207 }
208 if (objectName.startsWith("HoSupSec_")) {
209 return QString("伸缩臂模块-%1").arg(objectName);
210 }
211 if (objectName.startsWith("EOAT_")) {
212 return QString("末端执行器-%1").arg(objectName);
213 }
214 if (objectName.startsWith("AGV_")) {
215 return QString("AGV小车-%1").arg(objectName);
216 }
217
218 // ==================== 动态词条处理 ====================
219 // 如果 newValue 包含 "Host unreachable" 等英文错误，这里无法直接在 mapControlName 处理，
220 // 因为这是值，不是控件名或操作名。
221 // 如果需要翻译 newValue，通常应在 UI 显示层的业务逻辑中进行。
222
223 // 如果还是没有匹配，返回原始名称
224 return objectName;
225}
