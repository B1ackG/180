187void AGVModbusManager::pollRegisters()
188{
189 if (!isConnected()) {
190 return;
191 }
192
193 // 分组轮询，避免一次读取过大导致阻塞，同时保证所有AGV UI所需地址都能被覆盖。
194 // 组0: 0 (OA/驻车相关控制位)
195 // 组1: 2 (底盘切换模式值)
196 // 组2: 50-51 (触边/避障/心跳位状态 + 驻车状态位)
197 // 组3: 102-105 (电池1、电池2、速度、点动位移)
198 // 组4: 110-113 (故障码1-4)
199 // 组5: 114-117 (故障码5-8)
200 // 组6: 500 (有线/无线控制模式)
201 static int pollGroup = 0;
202
203 switch (pollGroup) {
204 case 0:
205 readMultipleRegisters(0, 1);
206 break;
207 case 1:
208 readMultipleRegisters(2, 1);
209 break;
210 case 2:
211 readMultipleRegisters(50, 2);
212 break;
213 case 3:
214 readMultipleRegisters(102, 4);
215 break;
216 case 4:
217 readMultipleRegisters(110, 4);
218 break;
219 case 5:
220 readMultipleRegisters(114, 4);
221 break;
222 case 6:
223 readMultipleRegisters(500, 1);
224 break;
225 default:
226 break;
227 }
228
229 pollGroup = (pollGroup + 1) % 7;
230}

237QByteArray AGVModbusManager::createReadRequest(int startAddress, int count)
238{
239 QByteArray request;
240
241 // 事务标识符 (2字节)
242 request.append(static_cast<char>((m_transactionId >> 8) & 0xFF));
243 request.append(static_cast<char>(m_transactionId & 0xFF));
244
245 // 记录事务ID到起始地址的映射
246 m_transactionAddressMap[m_transactionId] = startAddress;
247 m_transactionId++;
248
249 // 协议标识符 (2字节) - Modbus = 0
250 request.append(static_cast<char>(0x00));
251 request.append(static_cast<char>(0x00));
252
253 // 长度 (2字节) - 后面字节数
254 int length = 6; // 单元标识符1 + 功能码1 + 起始地址2 + 寄存器数量2
255 request.append(static_cast<char>((length >> 8) & 0xFF));
256 request.append(static_cast<char>(length & 0xFF));
257
258 // 单元标识符 (1字节) - 从站ID
259 request.append(static_cast<char>(1)); // 默认从站ID为1
260
261 // 功能码 (1字节) - 0x03 读保持寄存器
262 request.append(static_cast<char>(0x03));
263
264 // 起始地址 (2字节)
265 request.append(static_cast<char>((startAddress >> 8) & 0xFF));
266 request.append(static_cast<char>(startAddress & 0xFF));
267
268 // 寄存器数量 (2字节)
269 request.append(static_cast<char>((count >> 8) & 0xFF));
270 request.append(static_cast<char>(count & 0xFF));
271
272 return request;
273}

275void AGVModbusManager::readMultipleRegisters(int startAddress, int count)
276{
277 if (!isConnected() || count <= 0 || count > 125) {
278 return;
279 }
280
281 // 清理过期的请求记录
282 QDateTime now = QDateTime::currentDateTime();
283 QMutableMapIterator<quint16, QDateTime> it(m_requestTimestamps);
284 while (it.hasNext()) {
285 it.next();
286 if (it.value().msecsTo(now) > REQUEST_TIMEOUT) {
287 qDebug() << "清理超时请求，事务ID:" << it.key();
288 m_transactionAddressMap.remove(it.key());
289 it.remove();
290 }
291 }
292
293 // 限制在途请求数量，避免网络异常时无限堆积
294 constexpr int kMaxPendingRequests = 32;
295 if (m_transactionAddressMap.size() >= kMaxPendingRequests) {
296 qWarning() << "AGV在途请求过多，跳过本次轮询发送。pending=" << m_transactionAddressMap.size();
297 return;
298 }
299
300 QByteArray request = createReadRequest(startAddress, count);
301
302 // 记录请求时间
303 quint16 requestId = m_transactionId - 1; // 注意：createReadRequest中已经增加了m_transactionId
304 m_requestTimestamps[requestId] = QDateTime::currentDateTime();
305
306 m_socket->write(request);
307}

320bool AGVModbusManager::parseResponse(QByteArray &data)
321{
322 bool parsedAny = false;
323 while (data.size() >= 7) {
324 quint16 length = (static_cast<quint8>(data[4]) << 8) | static_cast<quint8>(data[5]);
325 int frameLength = 6 + length;
326
327 if (frameLength < 9) {
328 qWarning() << "非法AGV Modbus帧长度:" << frameLength;
329 data.clear();
330 emit errorOccurred("非法AGV Modbus帧长度");
331 return false;
332 }
333
334 if (data.size() < frameLength) {
335 break;
336 }
337
338 if (!parseSingleResponseFrame(data)) {
339 qWarning() << "解析AGV Modbus响应帧失败";
340 return false;
341 }
342 parsedAny = true;
343 }
344 return parsedAny;
345}

347bool AGVModbusManager::parseSingleResponseFrame(QByteArray &data)
348{
349 if (data.size() < 9) {
350 return false;
351 }
352
353 // 提取事务ID
354 quint16 transactionId = (static_cast<quint8>(data[0]) << 8) | static_cast<quint8>(data[1]);
355
356 // 提取长度字段（偏移4-5字节）
357 quint16 length = (static_cast<quint8>(data[4]) << 8) | static_cast<quint8>(data[5]);
358
359 // 计算完整帧长度
360 int frameLength = 6 + length; // MBAP头6字节 + 长度字段的值
361
362 if (data.size() < frameLength) {
363 return false;
364 }
365
366 // 提取完整的帧
367 QByteArray frame = data.left(frameLength);
368 data.remove(0, frameLength); // 移除已处理的数据
369
370 // 解析这个帧
371 return processSingleResponseFrame(frame, transactionId);
372}

374bool AGVModbusManager::processSingleResponseFrame(const QByteArray &frame, quint16 transactionId)
375{
376
377 // 跳过MBAP头（7字节）
378 QByteArray pdu = frame.mid(7);
379
380 if (pdu.size() < 1) {
381 qWarning() << "PDU数据不足";
382 return false;
383 }
384
385 quint8 functionCode = static_cast<quint8>(pdu[0]);
386
387 // 检查异常响应
388 if (functionCode & 0x80) {
389 quint8 errorCode = static_cast<quint8>(pdu[1]);
390 QString errorMsg = QString("Modbus异常: 错误码 0x%1").arg(errorCode, 2, 16, QChar('0'));
391 qWarning() << errorMsg;
392 m_transactionAddressMap.remove(transactionId);
393 m_requestTimestamps.remove(transactionId);
394 emit errorOccurred(errorMsg);
395 return false;
396 }
397
398 // 处理功能码 0x03（读保持寄存器）响应
399 if (functionCode == 0x03) {
400 if (!m_transactionAddressMap.contains(transactionId)) {
401 // 请求已超时清理或收到过期/乱序响应，直接忽略该帧，避免错误映射到地址0。
402 m_requestTimestamps.remove(transactionId);
403 qWarning() << "收到无映射事务ID的读响应，已忽略 transactionId=" << transactionId;
404 return true;
405 }
406
407 int startAddress = m_transactionAddressMap.take(transactionId);
408 m_requestTimestamps.remove(transactionId);
409
410 // 字节数
411 quint8 byteCount = static_cast<quint8>(pdu[1]);
412
413 // 寄存器数量
414 int registerCount = byteCount / 2;
415
416 for (int i = 0; i < registerCount; i++) {
417 int dataIndex = 2 + i * 2;
418 if (dataIndex + 1 >= pdu.size()) {
419 qWarning() << "数据不足，无法解析寄存器";
420 break;
421 }
422
423 quint16 value = (static_cast<quint8>(pdu[dataIndex]) << 8) |
424 static_cast<quint8>(pdu[dataIndex + 1]);
425
426 int address = startAddress + i;
427
428 // 发射寄存器值变化信号
429 emit registerValueChanged(address, value);
430
431 // 根据地址范围处理数据
432 if (address >= 50 && address <= 51) {
433 // 位变量区域（50-51）
434 processBitVariables(address, value);
435 } else if (address >= 102 && address <= 117) {
436 // 这个区域既有位变量又有字变量
437 if (address == 102) {
438 // 地址102：先处理位变量（故障状态）
439 processBitVariables(address, value);
440 // 然后处理字变量（电池1电量）
441 processWordVariables(address, value);
442 } else if (address == 103) {
443 // 地址103：电池2电量
444 processWordVariables(address, value);
445 } else {
446 // 其他地址：直接处理字变量
447 processWordVariables(address, value);
448 }
449 }
450 }
451 return true;
452 }
453 // 添加功能码 0x06（写单个寄存器）响应处理
454 else if (functionCode == 0x06) {
455 m_transactionAddressMap.remove(transactionId);
456 m_requestTimestamps.remove(transactionId);
457
458 if (pdu.size() >= 5) {
459 // 提取写入的地址
460 int writtenAddress = (static_cast<quint8>(pdu[1]) << 8) |
461 static_cast<quint8>(pdu[2]);
462
463 // 提取写入的值
464 quint16 writtenValue = (static_cast<quint8>(pdu[3]) << 8) |
465 static_cast<quint8>(pdu[4]);
466
467 // 发出写入完成信号
468 emit writeCompleted(writtenAddress, writtenValue, true);
469
470 return true;
471 }
472 }
473 else {
474 qWarning() << "不支持的功能码: 0x" << QString::number(functionCode, 16).toUpper();
475 }
476
477 return false;
478}

480void AGVModbusManager::processBitVariables(int address, quint16 value)
481{
482 // 处理位变量（每个位对应一个BOOL变量）
483 quint16 oldValue = m_registerValues.contains(address) ? m_registerValues[address] : 0;
484
485 // 如果值没有变化，直接返回
486 if (oldValue == value) {
487 return;
488 }
489
490 // 更新寄存器值
491 m_registerValues[address] = value;
492
493 for (int bitPos = 0; bitPos < 16; bitPos++) {
494 bool bitValue = (value >> bitPos) & 0x01;
495 bool oldBitValue = (oldValue >> bitPos) & 0x01;
496
497 // 检查位状态是否变化
498 if (oldBitValue != bitValue) {
499 m_bitStates[address][bitPos] = bitValue;
500
501 // 发射位变量变化信号
502 emit bitVariableChanged(address, bitPos, bitValue);
503
504 // 处理特定位变量
505 QString bitName = getBitVariableName(address, bitPos);
506 if (!bitName.isEmpty()) {
507 // 心跳位不需要显示
508 if (bitName == "heartbeat") {
509 continue;
510 }
511
512 // 故障相关的位由updateFaultsDisplay统一处理
513 if (address == 102 && (bitPos == 0 || bitPos == 1 || bitPos == 2)) {
514 continue;
515 }
516
517 // 其他位变量更新对应的Label
518 QString displayText;
519 if (bitValue) {
520 if (bitName == "label_back_slow") displayText = "后避障减速触发";
521 else if (bitName == "label_front_stop") displayText = "前避障停止触发";
522 else if (bitName == "label_front_touch") displayText = "前触边触发";
523 else if (bitName == "label_front_slow") displayText = "前避障减速触发";
524 else if (bitName == "label_back_stop") displayText = "后避障停止触发";
525 else if (bitName == "label_back_touch") displayText = "后触边触发";
526 else if (bitName == "label_left_touch") displayText = "左触边触发";
527 else if (bitName == "label_right_touch") displayText = "右触边触发";
528 else displayText = "触发";
529 } else {
530 displayText = "无动作";
531 }
532
533 // 确保Label名称有正确的格式
534 QString labelName = bitName;
535 if (!labelName.startsWith("label_")) {
536 labelName = "label_" + bitName;
537 }
538
539 qDebug() << " 更新Label:" << labelName << "->" << displayText;
540 emit updateStatusLabel(labelName, displayText);
541
542 // 特别处理特定的位变量
543 if (bitName == "back_touch") {
544 qDebug() << " 后触边状态变化:" << (bitValue ? "触发" : "无动作");
545 }
546
547 if (bitName == "jog_running") {
548 qDebug() << " 点动运行状态变化:" << (bitValue ? "运行中" : "停止");
549 }
550 }
551
552 // 特别处理心跳位
553 if (address == 50 && bitPos == 0) {
554 if (bitValue && !m_lastHeartbeatState) {
555 m_lastHeartbeatTime = QDateTime::currentDateTime();
556 qDebug() << " 收到心跳信号，时间:" << m_lastHeartbeatTime.toString("hh:mm:ss.zzz");
557 emit heartbeatReceived();
558 }
559 m_lastHeartbeatState = bitValue;
560 }
561 }
562 }
563
564 // 更新故障显示（处理地址52的故障位）
565 if (address == 102) {
566 qDebug() << " 更新故障显示";
567 updateFaultsDisplay();
568 }
569
570 qDebug() << "=== 位变量处理完成 ===";
571}

573void AGVModbusManager::processWordVariables(int address, quint16 value)
574{
575 qDebug() << "=== 开始处理字变量 ===";
576 qDebug() << "【进入processWordVariables】地址:" << address << "值:" << value;
577
578 QString varName = getWordVariableName(address);
579 qDebug() << "获取变量名 - 地址:" << address << "->" << varName;
580
581 if (varName.isEmpty()) {
582 qDebug() << "变量名为空，跳过处理";
583 qDebug() << "=== 字变量处理完成 ===";
584 return;
585 }
586
587 if (varName == "battery1") {
588 // 电池1电量 (0-100%)
589 int batteryPercent = qMin(value, static_cast<quint16>(100));
590 qDebug() << "电池1电量:" << batteryPercent << "%";
591 qDebug() << "发出updateProgressBar信号: progressBar_battery1, " << batteryPercent;
592 emit updateProgressBar("progressBar_battery1", batteryPercent);
593 emit updateStatusLabel("label_battery1_text", QString("%1%").arg(batteryPercent));
594 emit wordVariableChanged(address, value); // 兜底链路：允许 UI 直接基于寄存器更新
595 }
596 else if (varName == "battery2") {
597 // 电池2电量 (0-100%)
598 int batteryPercent = qMin(value, static_cast<quint16>(100));
599 qDebug() << "电池2电量:" << batteryPercent << "%";
600 emit updateStatusLabel("label_battery2_text", QString("%1%").arg(batteryPercent));
601 emit wordVariableChanged(address, value); // 兜底链路：允许 UI 直接基于寄存器更新
602 }
603 else if (varName == "speed") {
604 // 行驶速度 (mm/s) - 更新到wordVariableChanged信号，由MainWindow处理
605 qDebug() << "行驶速度:" << value << "mm/s";
606 // 这里不需要单独处理，wordVariableChanged信号会触发MainWindow的处理
607 emit wordVariableChanged(address, value);
608 }
609 else if (varName == "jog_displacement") {
610 // 点动位移
611 qDebug() << "点动位移:" << value << "mm";
612 emit updateStatusLabel("label_jog_displacement", QString("%1 mm").arg(value));
613 emit wordVariableChanged(address, value);
614 }
615 else if (varName.startsWith("fault_code_")) {
616 // 故障代码处理
617 QString faultName = varName.mid(11); // 去掉"fault_code_"前缀
618 QString displayText = QString("%1: 代码 %2").arg(faultName).arg(value);
619
620 if (value != 0) {
621 // 有故障代码
622 m_faultCodes[address] = value;
623 qDebug() << "故障代码" << faultName << ":" << value;
624 qDebug() << "发出addFaultCodeToList信号:" << displayText;
625
626 // 发射信号到UI
627 emit addFaultCodeToList(displayText);
628
629 // 记录到日志
630 qDebug() << "发现故障: " << displayText;
631 } else {
632 // 故障代码为0，表示无故障
633 m_faultCodes.remove(address);
634 qDebug() << "故障代码" << faultName << "已清除";
635 }
636
637 // 更新故障代码显示
638 updateFaultCodesDisplay();
639 emit wordVariableChanged(address, value);
640 }
641 else {
642 qDebug() << "未知的字变量类型:" << varName;
643 emit wordVariableChanged(address, value);
644 }
645
646 qDebug() << "=== 字变量处理完成 ===";
647}

782bool AGVModbusManager::writeSingleRegister(int address, quint16 value)
783{
784 // 如果全局禁用了写操作，则直接阻止并返回失败（用于故障排查）
785 if (!m_writesEnabled) {
786 qWarning() << "AGV 写操作已被禁用，忽略写入请求 地址:" << address << "值:" << value;
787 return false;
788 }
789
790 if (!isConnected()) {
791 if (!m_disconnectedWriteWarnedAddresses.contains(address)) {
792 qWarning() << "AGV Modbus未连接，无法写入地址" << address;
793 m_disconnectedWriteWarnedAddresses.insert(address);
794 }
795 return false;
796 }
797
798 m_disconnectedWriteWarnedAddresses.remove(address);
799
800 QByteArray request = createWriteRequest(address, value);
801
802 // 记录请求时间
803 quint16 requestId = m_transactionId - 1;
804 m_requestTimestamps[requestId] = QDateTime::currentDateTime();
805
806 m_socket->write(request);
807
808 return true;
809}

811QByteArray AGVModbusManager::createWriteRequest(int address, quint16 value)
812{
813 QByteArray request;
814
815 // 事务标识符 (2字节)
816 request.append(static_cast<char>((m_transactionId >> 8) & 0xFF));
817 request.append(static_cast<char>(m_transactionId & 0xFF));
818
819 // 记录事务ID
820 m_transactionAddressMap[m_transactionId] = address;
821 m_transactionId++;
822
823 // 协议标识符 (2字节) - Modbus = 0
824 request.append(static_cast<char>(0x00));
825 request.append(static_cast<char>(0x00));
826
827 // 长度 (2字节) - 后面字节数
828 int length = 6; // 单元标识符1 + 功能码1 + 地址2 + 值2
829 request.append(static_cast<char>((length >> 8) & 0xFF));
830 request.append(static_cast<char>(length & 0xFF));
831
832 // 单元标识符 (1字节) - 从站ID
833 request.append(static_cast<char>(1)); // 默认从站ID为1
834
835 // 功能码 (1字节) - 0x06 写单个寄存器
836 request.append(static_cast<char>(0x06));
837
838 // 寄存器地址 (2字节)
839 request.append(static_cast<char>((address >> 8) & 0xFF));
840 request.append(static_cast<char>(address & 0xFF));
841
842 // 寄存器值 (2字节)
843 request.append(static_cast<char>((value >> 8) & 0xFF));
844 request.append(static_cast<char>(value & 0xFF));
845
846 qDebug() << "AGV写入请求 - 地址:" << address
847 << "值:" << value
848 << "功能码: 0x06";
849
850 return request;
851}
