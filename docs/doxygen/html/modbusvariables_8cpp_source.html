108bool ModbusVariables::parseAddress(const QString &plcAddress, int &modbusAddress, int &bitPos)
109{
110 // 解析类似 %MX100.0 或 %MW102 的地址
111 if (plcAddress.startsWith("%MX")) {
112 // 位变量
113 QStringList parts = plcAddress.mid(3).split('.');
114 if (parts.size() == 2) {
115 modbusAddress = parts[0].toInt();
116 bitPos = parts[1].toInt();
117 return true;
118 }
119 } else if (plcAddress.startsWith("%MW")) {
120 // 字变量
121 modbusAddress = plcAddress.mid(3).toInt();
122 bitPos = -1; // 字变量没有位位置
123 return true;
124 }
125
126 return false;
127}

129QString ModbusVariables::getDisplayText(const ModbusVariable &var, quint16 value)
130{
131 switch (var.type) {
132 case VAR_BOOL:
133 // 对于位变量，我们假设传入的value已经是该位的值（0或1）
134 return value ? "触发" : "未触发";
135
136 case VAR_INT:
137 if (!var.unit.isEmpty()) {
138 return QString("%1 %2").arg(static_cast<qint16>(value)).arg(var.unit);
139 }
140 return QString::number(static_cast<qint16>(value));
141
142 case VAR_UINT:
143 if (!var.unit.isEmpty()) {
144 return QString("%1 %2").arg(value).arg(var.unit);
145 }
146 return QString::number(value);
147
148 default:
149 return QString::number(value);
150 }
151}
