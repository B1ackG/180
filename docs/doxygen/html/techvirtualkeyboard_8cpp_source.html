324void TechVirtualKeyboard::showAtWidget(QWidget *targetWidget)
325{
326 if (!targetWidget) return;
327
328 // 计算显示位置（目标控件下方）
329 QPoint globalPos = targetWidget->mapToGlobal(QPoint(0, targetWidget->height() + 5));
330
331 // 确保键盘不会超出屏幕
332 QScreen *screen = QGuiApplication::screenAt(globalPos);
333 if (!screen) {
334 screen = QGuiApplication::primaryScreen();
335 }
336
337 QRect screenRect = screen->availableGeometry();
338
339 // 边界检查
340 if (globalPos.x() + width() > screenRect.right()) {
341 globalPos.setX(screenRect.right() - width());
342 }
343 if (globalPos.y() + height() > screenRect.bottom()) {
344 globalPos.setY(targetWidget->mapToGlobal(QPoint(0, 0)).y() - height() - 5);
345 }
346 if (globalPos.x() < screenRect.left()) {
347 globalPos.setX(screenRect.left());
348 }
349
350 move(globalPos);
351 show();
352 raise();
353 activateWindow();
354
355 // 清空预览框内容
356 if (m_targetLineEdit) {
357 m_currentText = m_targetLineEdit->text();
358 } else {
359 m_currentText.clear();
360 }
361 updatePreview();
362}

364bool TechVirtualKeyboard::eventFilter(QObject *watched, QEvent *event)
365{
366 // 处理鼠标按下事件
367 if (event->type() == QEvent::MouseButtonPress) {
368 QMouseEvent *mouseEvent = static_cast<QMouseEvent*>(event);
369 QPoint globalPos = mouseEvent->globalPos();
370
371 // 检查点击是否在键盘内部
372 if (!geometry().contains(globalPos) && isVisible()) {
373 // 点击键盘外部，取消输入
374 hide();
375 emit inputCancelled();
376 }
377 }
378
379 return QWidget::eventFilter(watched, event);
