173void TechSliderLabel::onSliderValueChanged(int sliderValue)
174{
175 double newValue = sliderToValue(sliderValue);

184 m_oldValue = m_value;

264void TechSliderLabel::setValue(double value)
265{
266 if (value < m_minimum) value = m_minimum;
267 if (value > m_maximum) value = m_maximum;
268
269 if (qAbs(m_value - value) > 0.0001) {
270 m_oldValue = m_value;
271 m_value = value;
272 updateSliderFromValue();
273
274 // 如果开启了力控模式，根据数值更新颜色
275 if (m_forceControlEnabled) {
276 double range = m_maximum - m_minimum;
277 if (range > 0) {
278 // 计算离极值的程度 (0-1), 0表示在中间, 1表示在最大或最小值
279 double center = (m_maximum + m_minimum) / 2.0;
280 double deviation = qAbs(m_value - center) / (range / 2.0);
281
282 // 限制在0-1范围
283 deviation = qBound(0.0, deviation, 1.0);
284
285 // 插值颜色: 原色 -> 红色 (#FF3333)
286 QColor targetRed(255, 51, 51);
287
288 auto interpolate = [](const QColor& start, const QColor& end, double t) {
289 return QColor(
290 start.red() + (end.red() - start.red()) * t,
291 start.green() + (end.green() - start.green()) * t,
292 start.blue() + (end.blue() - start.blue()) * t,
293 start.alpha() + (end.alpha() - start.alpha()) * t
294 );
295 };
296
297 m_primaryColor = interpolate(m_originalPrimaryColor, targetRed, deviation);
298 m_secondaryColor = interpolate(m_originalSecondaryColor, targetRed, deviation);
299 m_glowColor = interpolate(m_originalGlowColor, QColor(255, 51, 51, 180), deviation);
300
301 applyTechStyle();
302 }
303 }
304
305 updateValueLabelFromValue();
306 emit valueChanged(m_value);
307 }
308}

541void TechSliderLabel::requestRepaint()
542{
543 if (!m_repaintElapsed.isValid()) {
544 m_repaintElapsed.start();
545 update();
546 return;
547 }
548
549 const qint64 elapsed = m_repaintElapsed.elapsed();
550 if (elapsed >= m_minRepaintIntervalMs) {
551 m_repaintElapsed.restart();
552 update();
553 return;
554 }
555
556 if (m_repaintPending) {
557 return;
558 }
559
560 m_repaintPending = true;
561 const int remainMs = m_minRepaintIntervalMs - static_cast<int>(elapsed);
562 QTimer::singleShot(remainMs, this, [this]() {
563 m_repaintPending = false;
564 m_repaintElapsed.restart();
565 update();
566 });
