245void TechSpeedGauge::drawObstacleIndicators(QPainter &painter)
246{
247 painter.save();
248 painter.setRenderHint(QPainter::Antialiasing);
249
250 // 使用环形扇区，避免 drawPie 产生“从圆心到边缘”的径向线
251 qreal outerRadius = m_radius * 1.05;
252 qreal innerRadius = m_radius * 0.95;
253 QRectF outerRect(-outerRadius, -outerRadius, outerRadius * 2, outerRadius * 2);
254 QRectF innerRect(-innerRadius, -innerRadius, innerRadius * 2, innerRadius * 2);
255
256 auto drawSector = [&](bool active, qreal startAngle, qreal spanAngle) {
257 if (!active) return;
258
259 // 设置红色半透明渐变
260 QRadialGradient gradient(0, 0, outerRadius);
261 gradient.setColorAt(0.8, QColor(255, 0, 0, 0)); // 内部透明
262 gradient.setColorAt(0.95, QColor(255, 0, 0, 150)); // 边缘处最亮
263 gradient.setColorAt(1.0, QColor(255, 0, 0, 200)); // 最外圈稍微深一点
264
265 QPainterPath path;
266 path.arcMoveTo(outerRect, startAngle);
267 path.arcTo(outerRect, startAngle, spanAngle);
268 path.arcTo(innerRect, startAngle + spanAngle, -spanAngle);
269 path.closeSubpath();
270
271 painter.setBrush(gradient);
272 painter.setPen(Qt::NoPen);
273 painter.drawPath(path);
274
275 // 外弧发光边框
276 QPainterPath outerArcPath;
277 outerArcPath.arcMoveTo(outerRect, startAngle);
278 outerArcPath.arcTo(outerRect, startAngle, spanAngle);
279 painter.setBrush(Qt::NoBrush);
280 painter.setPen(QPen(QColor(255, 50, 50, 200), 2));
281 painter.drawPath(outerArcPath);
282 };
283
284 painter.translate(m_center);
285
286 // 1. 前方 (Top/90度附近，Qt中是90度左右)
287 drawSector(m_obstacleFront, 45, 90);
288 // 2. 后方 (Bottom/270度附近)
289 drawSector(m_obstacleBack, 225, 90);
290 // 3. 左侧 (Left/180度附近)
291 drawSector(m_obstacleLeft, 135, 90);
292 // 4. 右侧 (Right/0度附近)
293 drawSector(m_obstacleRight, 315, 90);
294
295 painter.restore();
296}

355void TechSpeedGauge::drawBackground(QPainter &painter)
356{
357 QPainterPath backgroundPath;
358 backgroundPath.addEllipse(m_center, m_radius, m_radius);
359
360 // 根据样式绘制不同背景
361 switch (m_style) {
362 case StyleCyberFuturistic:
363 {
364 // 赛博风格：深色渐变背景
365 QRadialGradient gradient(m_center, m_radius);
366 gradient.setColorAt(0, QColor(20, 25, 45, 200));
367 gradient.setColorAt(0.7, QColor(10, 15, 30, 180));
368 gradient.setColorAt(1, QColor(5, 10, 20, 150));
369
370 painter.fillPath(backgroundPath, QBrush(gradient));
371
372 // 绘制网格线
373 painter.save();
374 painter.setPen(QPen(QColor(50, 60, 80, 80), 1));
375 for (int i = 0; i < 36; i++) {
376 qreal angle = i * 10.0;
377 QPointF p1 = pointOnCircle(m_radius * 0.9, angle);
378 QPointF p2 = pointOnCircle(m_radius * 0.7, angle);
379 painter.drawLine(m_center + p1, m_center + p2);
380 }
381 painter.restore();
382 break;
383 }
384
385 case StyleEnergyArc:
386 {
387 // 能量风格：深色到亮色的径向渐变
388 QRadialGradient gradient(m_center, m_radius);
389 gradient.setColorAt(0, QColor(30, 15, 10, 220));
390 gradient.setColorAt(0.5, QColor(20, 10, 5, 180));
391 gradient.setColorAt(1, QColor(10, 5, 0, 150));
392
393 painter.fillPath(backgroundPath, QBrush(gradient));
394 break;
395 }
396
397 case StyleHolographic:
398 {
399 // 全息风格：多层半透明圆环
400 painter.setBrush(Qt::NoBrush);
401 for (int i = 0; i < 4; i++) {
402 qreal r = m_radius - i * 5;
403 QColor color = m_primaryColor;
404 color.setAlpha(30 - i * 5);
405
406 painter.setPen(QPen(color, 1.5, Qt::DashLine));
407 painter.drawEllipse(m_center, r, r);
408 }
409 break;
410 }
411
412 case StyleClassicDial:
413 {
414 // 经典风格：光滑渐变
415 QRadialGradient gradient(m_center, m_radius);
416 gradient.setColorAt(0, QColor(40, 40, 50));
417 gradient.setColorAt(1, QColor(20, 20, 30));
418
419 painter.fillPath(backgroundPath, QBrush(gradient));
420
421 // 添加金属边框
422 QLinearGradient borderGrad(m_center - QPointF(m_radius, 0),
423 m_center + QPointF(m_radius, 0));
424 borderGrad.setColorAt(0, QColor(100, 100, 120));
425 borderGrad.setColorAt(0.5, QColor(180, 180, 200));
426 borderGrad.setColorAt(1, QColor(100, 100, 120));
427
428 painter.setPen(QPen(QBrush(borderGrad), 3));
429 painter.setBrush(Qt::NoBrush);
430 painter.drawEllipse(m_center, m_radius, m_radius);
431 break;
432 }
433 }

497// 绘制数值弧
498void TechSpeedGauge::drawValueArc(QPainter &painter)
499{
500 painter.save();
501
502 // 计算当前值对应的角度
503 qreal currentAngle = angleFromValue(m_currentValue);
504 qreal startAngle = 150.0; // 起始角度
505 qreal spanAngle = startAngle - currentAngle;
506
507 // 创建弧形路径
508 QPainterPath arcPath;
509 arcPath.arcMoveTo(m_gaugeRect, startAngle);
510 arcPath.arcTo(m_gaugeRect, startAngle, -spanAngle);
511
512 // 根据样式设置弧线样式
513 switch (m_style) {
514 case StyleCyberFuturistic:
515 {
516 // 赛博风格：锥形渐变
517 QConicalGradient gradient(m_center, -startAngle);
518 gradient.setColorAt(0.0, m_primaryColor);
519 gradient.setColorAt(0.5, m_secondaryColor);
520 gradient.setColorAt(1.0, m_primaryColor);
521
522 QPen arcPen(QBrush(gradient), 12);
523 arcPen.setCapStyle(Qt::FlatCap);
524 painter.setPen(arcPen);
525 painter.setBrush(Qt::NoBrush);
526 painter.drawPath(arcPath);
527 break;
528 }
529
530 case StyleEnergyArc:
531 {
532 // 能量风格：实心弧带
533 QPainterPath thickArc;
534 thickArc.arcMoveTo(m_gaugeRect.adjusted(-8, -8, 8, 8), startAngle);
535 thickArc.arcTo(m_gaugeRect.adjusted(-8, -8, 8, 8), startAngle, -spanAngle);
536
537 QLinearGradient gradient(m_center - QPointF(m_radius, 0),
538 m_center + QPointF(m_radius, 0));
539 gradient.setColorAt(0.0, m_primaryColor);
540 gradient.setColorAt(0.5, m_secondaryColor);
541 gradient.setColorAt(1.0, m_primaryColor);
542
543 painter.setPen(Qt::NoPen);
544 painter.setBrush(QBrush(gradient));
545 painter.drawPath(thickArc);
546 break;
547 }
548
549 case StyleHolographic:
550 {
551 // 全息风格：多层半透明弧线
552 for (int i = 0; i < 3; i++) {
553 QPen arcPen(m_primaryColor, 4 - i);
554 arcPen.setStyle(Qt::DotLine);
555 arcPen.setCapStyle(Qt::RoundCap);
556 painter.setPen(arcPen);
557 painter.drawPath(arcPath);
558 }
559 break;
560 }
561
562 case StyleClassicDial:
563 {
564 // 经典风格：光滑渐变弧线
565 QPen arcPen(QBrush(m_primaryColor), 10);
566 arcPen.setCapStyle(Qt::RoundCap);
567 painter.setPen(arcPen);
568 painter.drawPath(arcPath);
569 break;
570 }
571 }
572
573 // 添加发光效果
574 if (m_glowEnabled) {
575 painter.setPen(QPen(m_glowColor, 2));
576 painter.setBrush(Qt::NoBrush);
577 painter.drawPath(arcPath);
578 }
579
580 painter.restore();
581}

626void TechSpeedGauge::drawValueDisplay(QPainter &painter)
627{
628 painter.save();
629
630 // 显示背景圆（可以调整大小，让数值区域更大）
631 qreal displayRadius = m_radius * 0.4; // 从0.3改为0.4，增大显示区域
632 QRadialGradient bgGrad(m_center, displayRadius);
633 bgGrad.setColorAt(0, QColor(255, 255, 255, 50));
634 bgGrad.setColorAt(1, QColor(100, 100, 120, 100));
635
636 painter.setBrush(QBrush(bgGrad));
637 painter.setPen(QPen(QColor(200, 220, 255, 150), 2));
638 painter.drawEllipse(m_center, displayRadius, displayRadius);
639
640 // 显示当前值（增大字体）
641 QFont valueFont = painter.font();
642 valueFont.setPointSize(24); // 从12增大到24
643 valueFont.setBold(true);
644 valueFont.setFamily("Microsoft YaHei, Segoe UI");
645 painter.setFont(valueFont);
646
647 QString valueText = QString::number(m_currentValue, 'f', m_precision);
648 QRectF valueRect(m_center.x() - displayRadius * 0.8,
649 m_center.y() - displayRadius * 0.8, // 调整位置
650 displayRadius * 1.6,
651 displayRadius * 1.0);
652
653 // 数值颜色渐变
654 QLinearGradient textGrad(valueRect.topLeft(), valueRect.bottomLeft());
655 textGrad.setColorAt(0, Qt::white);
656 textGrad.setColorAt(1, m_primaryColor);
657
658 painter.setPen(QPen(QBrush(textGrad), 2)); // 加粗线条
659 painter.drawText(valueRect, Qt::AlignCenter, valueText);
660
661 // 显示单位（调整位置，现在数值变大，单位需要下移）
662 QFont unitFont = painter.font();
663 unitFont.setPointSize(14); // 稍微增大单位字体
664 painter.setFont(unitFont);
