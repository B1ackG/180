118void FeatureSwitchManager::ensureConfigExists()
119{
120 // 确保配置文件存在；若不存在则创建默认配置并开启所有功能。
121 // 这样首次运行时会生成一个可编辑的 INI 文件，方便用户修改开关。
122 const QString path = configFilePath();
123 if (QFileInfo::exists(path)) {
124 return;
125 }
126
127 QSettings s(path, QSettings::IniFormat);
128 // 全局主开关，默认开启
129 s.setValue("Master/enabled", true);
130
131 // 写入所有大功能节点，默认值为 true（开启）
132 s.beginGroup("BigFeatures");
133 for (const QString &key : m_allBigFeatures) {
134 s.setValue(key, true);
135 }
136 s.endGroup();
137
138 // 写入所有小功能节点，默认值为 true（开启）
139 s.beginGroup("SmallFeatures");
140 for (const QString &key : m_allSmallFeatures) {
141 // 特殊处理：模拟器模式默认关闭
142 if (key == "tcp.local_simulator" || key == "tcp.remote_simulator") {
143 s.setValue(key, false);
144 } else {
145 s.setValue(key, true);
146 }
147 }
148 s.endGroup();
149
150 s.sync();
151}

153void FeatureSwitchManager::reload()
154{
155 // 从配置文件读取当前所有开关状态并在内存中建立“禁用集合”。
156 // 保存的是被禁用的项集合（m_disabledBigFeatures / m_disabledSmallFeatures），
157 // 这样查询时可以通过判断集合是否包含 key 来决定是否关闭功能。
158 const QString path = configFilePath();
159 QSettings s(path, QSettings::IniFormat);
160
161 m_masterEnabled = s.value("Master/enabled", true).toBool();
162 m_disabledBigFeatures.clear();
163 m_disabledSmallFeatures.clear();
164
165 s.beginGroup("BigFeatures");
166 for (const QString &key : m_allBigFeatures) {
167 // 如果配置中该条目为 false，则加入禁用集合
168 if (!s.value(key, true).toBool()) {
169 m_disabledBigFeatures.insert(key);
170 }
171 }
172 s.endGroup();
173
174 s.beginGroup("SmallFeatures");
175 for (const QString &key : m_allSmallFeatures) {
176 if (!s.value(key, true).toBool()) {
177 m_disabledSmallFeatures.insert(key);
178 }
179 }
180 s.endGroup();
181}

183void FeatureSwitchManager::save()
184{
185 // 将当前内存中的状态写回配置文件。注意我们在内存中保存的是“被禁用”的集合，
186 // 写回时要把布尔值取反（存在于禁用集合 -> 写 false；否则写 true）。
187 const QString path = configFilePath();
188 QSettings s(path, QSettings::IniFormat);
189
190 s.setValue("Master/enabled", m_masterEnabled);
191
192 s.beginGroup("BigFeatures");
193 for (const QString &key : m_allBigFeatures) {
194 s.setValue(key, !m_disabledBigFeatures.contains(key));
195 }
196 s.endGroup();
197
198 s.beginGroup("SmallFeatures");
199 for (const QString &key : m_allSmallFeatures) {
200 s.setValue(key, !m_disabledSmallFeatures.contains(key));
201 }
202 s.endGroup();
203
204 s.sync();
205}

212bool FeatureSwitchManager::isBigFeatureEnabled(const QString &bigKey) const
213{
214 // 首先检查主开关，如果主开关关闭则所有功能都不可用。
215 if (!m_masterEnabled) {
216 return false;
217 }
218
219 // 如果传入的 key 并不在已知的大功能集合中，默认认为该功能可用（避免阻塞未知/新增功能）。
220 if (!m_allBigFeatures.contains(bigKey)) {
221 return true;
222 }
223
224 // 返回该大功能是否未被标记为禁用。
225 return !m_disabledBigFeatures.contains(bigKey);
226}

228bool FeatureSwitchManager::isSmallFeatureEnabled(const QString &smallKey) const
229{
230 // 同上：检查主开关
231 if (!m_masterEnabled) {
232 return false;

234
235 // 若该小功能不是已知列表的一部分，则默认允许（不阻塞未知项）。

237 return true;

243bool FeatureSwitchManager::isFeatureEnabled(const QString &bigKey, const QString &smallKey) const
244{
245 // 组合查询：先检查所属大功能，若大功能被禁用则直接返回 false。
246 if (!isBigFeatureEnabled(bigKey)) {
247 return false;
248 }
249
250 // 如果没有提供小功能 key，则仅依赖大功能的开关状态。
251 if (smallKey.isEmpty()) {
252 return true;
253 }
254
255 // 否则还需要检查小功能的开关状态。
256 return isSmallFeatureEnabled(smallKey);
