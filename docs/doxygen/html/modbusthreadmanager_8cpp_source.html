66void ModbusThreadManager::registerSlider(TechSliderEdit *slider, int address)
67{
68 if (!slider || address < 0) {
69 return;
70 }
71
72 if (m_sliderToAddress.contains(slider)) {
73 int oldAddress = m_sliderToAddress[slider];
74 m_addressToSlider.remove(oldAddress);
75 m_sliderToAddress.remove(slider);
76 if (m_modbusClient) {
77 m_modbusClient->removeRegisterFromPoll(oldAddress);

79 }
80
81 m_addressToSlider[address] = slider;
82 m_sliderToAddress[slider] = address;
83 slider->setModbusAddress(address);
84

259bool ModbusThreadManager::writeSingleRegister(int address, quint16 value)
260{
261 if (!m_modbusClient || !m_modbusClient->isConnected()) {
262 qWarning() << "Modbus客户端未连接，无法写入地址" << address;
263 emit writeOperationComplete(false, QString("Modbus未连接"));
264 return false;
265 }
266
267 bool result = m_modbusClient->writeSingleRegister(address, value);
268 if (result) {
269 emit registerWritten(address, value);
270 emit writeOperationComplete(true, QString("写入地址%1成功").arg(address));
271 } else {
272 emit writeOperationComplete(false, QString("写入地址%1失败").arg(address));
273 }
274 return result;

295void ModbusThreadManager::readMultipleHoldingRegisters(int startAddress, int count)
296{
297 if (!m_modbusClient || !m_modbusClient->isConnected()) {
298 qWarning() << "Modbus客户端未连接，无法批量读取保持寄存器";
299 return;
300 }
301
302 const int maxReadCount = 125;
303 for (int i = 0; i < count; i += maxReadCount) {
304 int currentStart = startAddress + i;
305 int currentCount = qMin(maxReadCount, count - i);
306 m_modbusClient->readHoldingRegisters(currentStart, currentCount);
307 }
