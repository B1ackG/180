48 // 投影计算函数
49 function project(v) {
50 var x = v.x, y = v.y, z = v.z;
51 var tmp;
52
53 // --- 1. 应用旋转 ---
54 // Rx (Roll - 绕Y轴)
55 tmp = x * Math.cos(root.roll) + z * Math.sin(root.roll);
56 z = -x * Math.sin(root.roll) + z * Math.cos(root.roll);
57 x = tmp;
58
59 // Ry (Pitch - 绕X轴)
60 tmp = y * Math.cos(root.pitch) - z * Math.sin(root.pitch);
61 z = y * Math.sin(root.pitch) + z * Math.cos(root.pitch);
62 y = tmp;
63
64 // Rz (Yaw - 绕Z轴)
65 tmp = x * Math.cos(root.yaw) - y * Math.sin(root.yaw);
66 y = x * Math.sin(root.yaw) + y * Math.cos(root.yaw);
67 x = tmp;
68
69 // --- 2. 应用位移 ---
70 x += root.xPos * 0.1; // X 左右
71 y += root.yPos * 0.1; // Y 前后
72 z += root.zPos * 0.1; // Z 垂直
73
74 // --- 3. 透视投影 ---
75 var viewDist = 300;
76 var depth = viewDist - y;
77 var scale = viewDist / (depth > 10 ? depth : 10);
78
79 return {
80 x: centerX + x * scale,
81 y: centerY - z * scale // 屏幕Y反向
82 };
83 }
