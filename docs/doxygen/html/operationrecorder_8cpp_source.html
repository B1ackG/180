60void OperationRecorder::loadSecuritySettings()
61{
62 QSettings settings("config.ini", QSettings::IniFormat);
63 settings.beginGroup("OperationLogTransport");
64
65 const QByteArray tlsEnv = qgetenv("AGV_LOG_TLS");
66 if (!tlsEnv.isEmpty()) {
67 const QByteArray normalized = tlsEnv.trimmed().toLower();
68 m_tcpUseTls = (normalized == "1" || normalized == "true" || normalized == "on");
69 } else {
70 m_tcpUseTls = settings.value("tls_enabled", true).toBool();
71 }
72
73 const QByteArray allowedHostsEnv = qgetenv("AGV_LOG_ALLOWED_HOSTS");
74 if (!allowedHostsEnv.isEmpty()) {
75 m_allowedHosts = parseCsvList(QString::fromUtf8(allowedHostsEnv));
76 } else {
77 m_allowedHosts = parseCsvList(settings.value("allowed_hosts", "127.0.0.1,192.168.1.70").toString());
78 }
79
80 const QByteArray signKeyEnv = qgetenv("AGV_LOG_SIGNING_KEY");
81 if (!signKeyEnv.isEmpty()) {
82 m_signingKey = signKeyEnv;
83 } else {
84 m_signingKey = settings.value("signing_key", "").toByteArray();
85 }
86
87#ifndef QT_NO_SSL
88 if (QSslSocket *sslSocket = qobject_cast<QSslSocket *>(m_tcpSocket)) {
89 QSslConfiguration sslConfig = sslSocket->sslConfiguration();
90 sslConfig.setPeerVerifyMode(QSslSocket::VerifyPeer);
91
92 const QByteArray caEnv = qgetenv("AGV_LOG_TLS_CA_FILE");
93 const QString caPath = caEnv.isEmpty()
94 ? settings.value("tls_ca_file", "").toString()
95 : QString::fromUtf8(caEnv);
96 if (!caPath.trimmed().isEmpty()) {
97 QFile caFile(caPath.trimmed());
98 if (caFile.open(QIODevice::ReadOnly)) {
99 const QList<QSslCertificate> certs = QSslCertificate::fromData(caFile.readAll(), QSsl::Pem);
100 if (!certs.isEmpty()) {
101 sslConfig.setCaCertificates(certs);
102 }
103 }
104 }
105
106 sslSocket->setSslConfiguration(sslConfig);
107 } else {
108 // 理论上仅在运行时SSL类型不匹配时触发，回退纯TCP保证可用性。
109 m_tcpUseTls = false;
110 }
111#else
112 // 目标Qt未启用SSL时，自动回退纯TCP。
113 m_tcpUseTls = false;
114#endif
115 settings.endGroup();
116}

118bool OperationRecorder::validateTransportPolicy(QString *reason) const
119{
120 constexpr int kMinSigningKeyLength = 16;
121
122 if (!m_allowedHosts.isEmpty() && !m_allowedHosts.contains(m_tcpServerIp)) {
123 if (reason) {
124 *reason = QString("目标主机不在白名单中: %1").arg(m_tcpServerIp);
125 }
126 return false;
127 }
128
129 if (!m_tcpUseTls && m_signingKey.isEmpty()) {
130 if (reason) {
131 *reason = QStringLiteral("TLS关闭时必须配置签名密钥(AGV_LOG_SIGNING_KEY)");
132 }
133 return false;
134 }
135
136 if (!m_tcpUseTls && m_signingKey.size() < kMinSigningKeyLength) {
137 if (reason) {
138 *reason = QStringLiteral("签名密钥长度过短，至少需要16字节");
139 }
140 return false;
141 }
142
143 return true;
144}

146QByteArray OperationRecorder::buildSignedPayload(const OperationRecord &record) const
147{
148 const QJsonObject payloadObj = record.toJson();
149 const QJsonDocument payloadDoc(payloadObj);
150 const QByteArray payload = payloadDoc.toJson(QJsonDocument::Compact);
151 const qint64 ts = QDateTime::currentMSecsSinceEpoch();
152 const QByteArray nonce = QCryptographicHash::hash(
153 QByteArray::number(ts) + payload,
154 QCryptographicHash::Sha256).toHex().left(16);
155
156 QJsonObject envelope;
157 envelope["payload"] = QString::fromUtf8(payload.toBase64());
158 envelope["ts"] = QString::number(ts);
159 envelope["nonce"] = QString::fromUtf8(nonce);
160 envelope["alg"] = "HMAC-SHA256";
161
162 if (!m_signingKey.isEmpty()) {
163 const QByteArray canonical = payload + '|' + QByteArray::number(ts) + '|' + nonce;
164 const QByteArray sig = QMessageAuthenticationCode::hash(canonical, m_signingKey, QCryptographicHash::Sha256).toHex();
165 envelope["sig"] = QString::fromUtf8(sig);
166 }
167
168 QJsonDocument envelopeDoc(envelope);
169 QByteArray out = envelopeDoc.toJson(QJsonDocument::Compact);
170 out.append("\n");
171 return out;
172}

356void OperationRecorder::sendRecordToServer(const OperationRecord &record)
357{
358 if (!m_tcpEnabled) {
359 return;
360 }
361
362 if (!isTcpConnected()) {
363 // 如果未连接，尝试连接
364 connectTcpSocket();
365
366 // 传输策略阻断时不再累积离线队列，避免UI长期运行后卡顿。
367 if (m_transportPolicyBlocked) {
368 return;
369 }
370
371 // 将记录添加到队列，等待连接成功后再发送
372 enqueueRecordIfPossible(record);
373 return;
374 }
375
376 const QByteArray data = buildSignedPayload(record);
377
378 // 发送数据
379 qint64 bytesWritten = m_tcpSocket->write(data);
380
381 if (bytesWritten == -1) {
382 qWarning() << "发送数据失败:" << m_tcpSocket->errorString();
383 // 添加到队列等待重试
384 enqueueRecordIfPossible(record);
385 } else {
386 qDebug() << "发送记录到服务器:" << record.controlName << "操作:" << record.operation;
387 }
388}

390void OperationRecorder::sendQueuedRecords()
391{
392 if (m_tcpSendQueue.isEmpty() || !isTcpConnected()) {
393 if (m_tcpSendQueue.isEmpty()) {
394 emit tcpTransmissionComplete();
395 }
396 return;
397 }
398
399 // 每次发送最多10条记录，避免阻塞
400 int sendCount = qMin(10, m_tcpSendQueue.size());
401
402 for (int i = 0; i < sendCount; ++i) {
403 OperationRecord record = m_tcpSendQueue.takeFirst();
404 sendRecordToServer(record);
405 }
406
407 // 继续发送剩余记录
408 if (!m_tcpSendQueue.isEmpty()) {
409 QTimer::singleShot(30, this, &OperationRecorder::sendQueuedRecords);
410 } else {
411 emit tcpTransmissionComplete();
412 }
413}

415void OperationRecorder::connectTcpSocket()
416{
417 if (!m_tcpEnabled || m_tcpSocket->state() == QAbstractSocket::ConnectingState ||
418 m_tcpSocket->state() == QAbstractSocket::ConnectedState) {
419 return;
420 }
421
422 QString policyError;
423 if (!validateTransportPolicy(&policyError)) {
424 m_transportPolicyBlocked = true;
425
426 const qint64 nowMs = QDateTime::currentMSecsSinceEpoch();
427 const bool shouldReport = (policyError != m_lastTransportPolicyError)
428 || ((nowMs - m_lastTransportPolicyErrorMs) > 3000);
429
430 if (shouldReport) {
431 m_lastTransportPolicyError = policyError;
432 m_lastTransportPolicyErrorMs = nowMs;
433 qWarning() << "日志传输策略阻止连接:" << policyError;
434 emit tcpTransmissionError(policyError);
435 } else {
436 qDebug() << "日志传输策略阻止连接(节流):" << policyError;
437 }
438 return;
439 }
440
441 m_transportPolicyBlocked = false;
442
443 qDebug() << "连接日志服务器:" << m_tcpServerIp << ":" << m_tcpServerPort << "TLS:" << m_tcpUseTls;
444#ifndef QT_NO_SSL
445 if (m_tcpUseTls) {
446 QSslSocket *sslSocket = qobject_cast<QSslSocket*>(m_tcpSocket);
447 if (sslSocket) {
448 sslSocket->connectToHostEncrypted(m_tcpServerIp, m_tcpServerPort);
449 return;
450 }
451 }
452#endif
453 m_tcpSocket->connectToHost(m_tcpServerIp, m_tcpServerPort);
454}
