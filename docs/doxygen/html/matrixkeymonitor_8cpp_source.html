96void MatrixKeyMonitor::onSocketActivated(int socket)
97{
98 // 这个槽函数会在工作线程中执行
99 struct input_event ev;
100
101 // 读取所有可用的事件
102 while (true) {
103 ssize_t bytesRead = read(socket, &ev, sizeof(ev));
104
105 if (bytesRead != sizeof(ev)) {
106 if (bytesRead < 0) {
107 if (errno == EAGAIN || errno == EWOULDBLOCK) {
108 // 没有更多数据了
109 break;
110 }
111 qWarning() << "读取键盘事件失败:" << strerror(errno);
112 break;
113 }
114 break;
115 }
116
117 // 只处理按键事件
118 if (ev.type == EV_KEY) {
119 int keyNumber = mapKeyCodeToButtonNumber(ev.code);
120 bool pressed = (ev.value == 1);
121
122 if (keyNumber >= 0) {
123 qDebug() << "矩阵按键事件 - 键码:" << ev.code
124 << "-> 按钮" << keyNumber
125 << "状态:" << (pressed ? "按下" : "释放");
126
127 // 发出信号
128 emit keyPressed(keyNumber, pressed);
129 emit rawKeyEvent(ev.code, ev.value);
130 }
131 }
132 }
