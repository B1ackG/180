126void TechArcGauge::paintEvent(QPaintEvent *)
127{
128 QPainter painter(this);
129 painter.setRenderHint(QPainter::Antialiasing);
130
131 int size = qMin(width(), height()) - 10;
132 QRectF rect((width() - size) / 2, (height() - size) / 2, size, size);
133
134 // 绘制背景半环 (180度, 底部开口)
135 float startAngle = -225;
136 float spanAngle = 270;
137
138 QPen backPen;
139 backPen.setColor(QColor(50, 50, 50, 100));
140 backPen.setWidth(8);
141 backPen.setCapStyle(Qt::RoundCap);
142 painter.setPen(backPen);
143 painter.drawArc(rect.adjusted(10, 10, -10, -10), startAngle * 16, spanAngle * 16);
144
145 // 绘制当前值环
146 float ratio = (m_value - m_minimum) / (m_maximum - m_minimum);
147 float valueSpan = spanAngle * ratio;
148
149 QPen valuePen;
150 valuePen.setColor(m_primaryColor);
151 valuePen.setWidth(10);
152 valuePen.setCapStyle(Qt::RoundCap);
153
154 // 发光效果
155 painter.save();
156 QPen glowPen = valuePen;
157 glowPen.setColor(m_glowColor);
158 glowPen.setWidth(15);
159 painter.setPen(glowPen);
160 painter.drawArc(rect.adjusted(10, 10, -10, -10), startAngle * 16, valueSpan * 16);
161 painter.restore();
162
163 painter.setPen(valuePen);
164 painter.drawArc(rect.adjusted(10, 10, -10, -10), startAngle * 16, valueSpan * 16);
165
166 // 绘制刻度点 (装饰用的点阵)
167 painter.setPen(QPen(m_primaryColor.lighter(), 2));
168 for (int i = 0; i <= 10; ++i) {
169 float angle = startAngle + (spanAngle * i / 10.0);
170 float rad = qDegreesToRadians(-angle);
171 float r = size / 2.0 - 5;
172 float centerX = width() / 2.0;
173 float centerY = height() / 2.0;
174 painter.drawPoint(centerX + r * cos(rad), centerY + r * sin(rad));
175 }
176
177 // 绘制扫描线特效 (一个小亮点在环上滑动)
178 float scanAngle = startAngle + m_scanLinePhase * (spanAngle / 360.0);
179 float scanRad = qDegreesToRadians(-scanAngle);
180 float scanR = size / 2.0 - 10;
181 painter.setBrush(m_primaryColor);
182 painter.setPen(Qt::NoPen);
183 painter.drawEllipse(QPointF(width() / 2.0 + scanR * cos(scanRad),
184 height() / 2.0 + scanR * sin(scanRad)), 3, 3);
185
186 // --- 绘制第二数值（例如速度）的进度条（在内圈） ---
187 if (m_secondMaximum > 0) {
188 float innerRatio = qBound(0.0, m_secondValue / m_secondMaximum, 1.0);
189 float secondSpan = spanAngle * innerRatio;
190
191 QPen secondPen;
192 secondPen.setColor(QColor(255, 165, 0, 180)); // 科技橙 (Orange)
193 secondPen.setWidth(4);
194 secondPen.setCapStyle(Qt::RoundCap);
195
196 painter.setPen(secondPen);
197 painter.drawArc(rect.adjusted(25, 25, -25, -25), startAngle * 16, secondSpan * 16);
198 }
199
200 // 绘制文字
201 painter.setPen(Qt::white);
202 QFont font = painter.font();
203
204 // 当前主数值（长度/角度）
205 font.setPixelSize(size / 6);
206 font.setBold(true);
207 painter.setFont(font);
208 QString valueStr = QString::number(m_value, 'f', m_precision);
209 painter.drawText(rect.adjusted(0, -size/15, 0, -size/15), Qt::AlignCenter, valueStr);
210
211 // 主数值单位
212 font.setPixelSize(size / 15);
213 font.setBold(false);
214 painter.setFont(font);
215 painter.drawText(rect.adjusted(0, size/8, 0, size/8), Qt::AlignCenter, m_suffix);
216
217 // 绘制第二数值文字（速度）
218 if (!m_secondSuffix.isEmpty() || m_secondValue != 0) {
219 font.setPixelSize(size / 13); // 增大字体由 18 变为 15 (数值越小字体越大)
220 font.setBold(true); // 设置为加粗增加辨识度
221 painter.setFont(font);
222 painter.setPen(QColor(255, 120, 0)); // 科技感橙色
223 QString secondStr = QString("V: %1 %2").arg(QString::number(m_secondValue, 'f', 1)).arg(m_secondSuffix);
224 painter.drawText(rect.adjusted(0, size/3.8, 0, size/3.8), Qt::AlignCenter, secondStr); // 稍微下移位置以避免拥挤
225 }
226
227 // 参数名称 (在顶部)
228 font.setPixelSize(size / 10);
229 painter.setPen(Qt::white);
230 painter.setFont(font);
231 painter.drawText(rect.adjusted(0, -size/3.5, 0, -size/3.5), Qt::AlignCenter, m_labelText);
232
233 // Min / Max
234 font.setPixelSize(size / 12); // 增大字体由 20 变为 15
235 painter.setFont(font);
236 painter.setPen(QColor(180, 180, 180));
237
238 // 计算圆弧起点 (-225度) 和 终点 (45度) 的坐标
239 // 注意：Qt 的 drawArc 使用的角度单位是 1/16 度，且顺时针为负，我们计算位置使用弧度
240 float rText = size / 2.0 - 5; // 稍微靠外一点
241 float centerX = width() / 2.0;
242 float centerY = height() / 2.0;
243
244 // 起点 (Min)
245 float minRad = qDegreesToRadians(225.0f); // -(-225)
246 QPointF minPos(centerX + rText * cos(minRad), centerY + rText * sin(minRad));
247
248 // 终点 (Max)
249 float maxRad = qDegreesToRadians(-45.0f); // -(45)
250 QPointF maxPos(centerX + rText * cos(maxRad), centerY + rText * sin(maxRad));
251
252 // 绘制文本，稍微偏移以避免覆盖圆弧
253 painter.drawText(QRectF(minPos.x() - 40, minPos.y() - 20, 40, 20), Qt::AlignRight | Qt::AlignVCenter, QString::number(m_minimum));
254 painter.drawText(QRectF(maxPos.x(), maxPos.y() - 20, 40, 20), Qt::AlignLeft | Qt::AlignVCenter, QString::number(m_maximum));
255}
