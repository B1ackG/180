298void TechSliderEdit::onLineEditEditingFinished()
299{
300 QString text = m_lineEdit->text();
301 if (text.isEmpty()) {
302 m_lineEdit->setText(QString::number(m_value, 'f', m_precision));
303 return;
304 }
305
306 bool ok;
307 double newValue = text.toDouble(&ok);
308
309 if (ok) {
310 // 限制在范围内
311 if (newValue < m_minimum) newValue = m_minimum;
312 if (newValue > m_maximum) newValue = m_maximum;
313 if (qAbs(newValue - m_value) > 0.0001) {
314 m_oldValue = m_value; // 保存旧值
315
316 setValue(newValue);
317 emit valueChangedWithRecord(m_oldValue, m_value); // 发出带记录的信号
318 }
319 emit editingFinished();
320 } else {
321 // 恢复原值
322 m_lineEdit->setText(QString::number(m_value, 'f', m_precision));
323 }
324}

326void TechSliderEdit::onPresetButtonClicked()
327{
328 QPushButton *btn = qobject_cast<QPushButton *>(sender());
329 if (!btn) return;
330
331 double ratio = 0.0;
332 if (btn == m_lowBtn) ratio = 0.1;
333 else if (btn == m_midBtn) ratio = 0.4;
334 else if (btn == m_highBtn) ratio = 0.8;
335
336 double newValue = m_maximum * ratio;
337
338 // 如果有精度要求，进行四舍五入
339 if (m_precision > 0) {
340 newValue = qRound(newValue * m_conversionFactor) / (double)m_conversionFactor;
341 } else {
342 newValue = qRound(newValue);
343 }
344
345 if (qAbs(newValue - m_value) > 0.0001) {
346 m_oldValue = m_value;
347 setValue(newValue);
348 emit valueChangedWithRecord(m_oldValue, m_value);
349 }
350 emit editingFinished();
351}

353void TechSliderEdit::onSliderValueChanged(int sliderValue)

363 m_oldValue = m_value; // 保存旧值

365 updateLineEditFromValue();
